<!-- Méthodes pédagogiques spécifiques -->
<category>
    <pattern>WHAT IS FLIPPED CLASSROOM</pattern>
    <template>It's a method where students learn new content at home via videos or readings and use class time for exercises and discussions.</template>
</category>

<category>
    <pattern>WHAT IS PROJECT BASED LEARNING</pattern>
    <template>Students gain knowledge and skills by working for an extended period to investigate and respond to a complex question or challenge.</template>
</category>

<category>
    <pattern>WHAT IS INQUIRY BASED LEARNING</pattern>
    <template>A method where learning begins with questions, problems, or scenarios rather than simply presenting facts.</template>
</category>

<category>
    <pattern>WHAT IS BLENDED LEARNING</pattern>
    <template>An approach that combines online educational materials and opportunities with traditional place-based classroom methods.</template>
</category>

<category>
    <pattern>WHAT IS GAMIFICATION IN EDUCATION</pattern>
    <template>Using game design elements like points, badges, and leaderboards in learning environments to motivate students.</template>
</category>

<category>
    <pattern>WHAT IS COOPERATIVE LEARNING</pattern>
    <template>Students work together in small groups to complete tasks and achieve shared learning goals.</template>
</category>

<category>
    <pattern>WHAT IS DIFFERENTIATED INSTRUCTION</pattern>
    <template>Tailoring instruction to meet individual student needs, interests, and readiness levels.</template>
</category>

<category>
    <pattern>WHAT IS MONTESSORI EDUCATION</pattern>
    <template>A child-centered approach that emphasizes independence, freedom within limits, and respect for a child's natural development.</template>
</category>

<category>
    <pattern>WHAT IS THE FREINET METHOD</pattern>
    <template>A student-centered approach focusing on experiential learning, cooperation, and democratic classroom practices.</template>
</category>

<category>
    <pattern>WHAT IS CRITICAL PEDAGOGY</pattern>
    <template>An approach that encourages students to question and challenge dominant beliefs and practices, especially related to power and justice.</template>
</category>

<category>
    <pattern>WHAT IS EXPERIENTIAL LEARNING</pattern>
    <template>Learning through reflection on doing, such as through hands-on activities, experiments, or real-world projects.</template>
</category>

<category>
    <pattern>WHAT IS DESIGN THINKING IN EDUCATION</pattern>
    <template>A creative problem-solving process used to develop innovative solutions in the classroom.</template>
</category>

<category>
    <pattern>WHAT IS SCAFFOLDING IN TEACHING</pattern>
    <template>Providing temporary support to help students learn new concepts, gradually removing aid as they gain mastery.</template>
</category>
